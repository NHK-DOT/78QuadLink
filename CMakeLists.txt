cmake_minimum_required(VERSION 3.16)
project(quadlink78 VERSION 0.5.1 LANGUAGES CXX)
include(GNUInstallDirs)
add_library(quadlink78 INTERFACE)
add_library(quadlink78::board ALIAS quadlink78)
set_target_properties(quadlink78 PROPERTIES EXPORT_NAME board)
target_compile_features(quadlink78 INTERFACE cxx_std_17)
target_include_directories(quadlink78 INTERFACE $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
include(CTest)
if(BUILD_TESTING)
 add_executable(board_roundtrip examples/board_roundtrip.cpp)
 target_link_libraries(board_roundtrip PRIVATE quadlink78::board)
 add_test(NAME board_cross_process COMMAND board_roundtrip)
endif()
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(TARGETS quadlink78 EXPORT quadlink78Targets)
install(EXPORT quadlink78Targets NAMESPACE quadlink78:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/quadlink78)
include(CMakePackageConfigHelpers)
configure_package_config_file(cmake/quadlink78Config.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/quadlink78Config.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/quadlink78)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/quadlink78ConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/quadlink78Config.cmake ${CMAKE_CURRENT_BINARY_DIR}/quadlink78ConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/quadlink78)
